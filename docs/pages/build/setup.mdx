import React, { useState } from 'react';
import { StyleSheet, Text, View, Dimensions, FlatList, TouchableOpacity, SafeAreaView, StatusBar, TextInput, Modal, Share, ScrollView, Platform, Alert } from 'react-native';
import * as DocumentPicker from 'expo-document-picker';
import * as MediaLibrary from 'expo-media-library';

const { height, width } = Dimensions.get('window');

const VIDEO_FEED = [
  { id: '1', title: 'Trending Indie Track 🎵', user: '@creator_one', likes: 12400, comments: 340 },
  { id: '2', title: 'Viral Synth-Pop 🚀', user: '@music_hustler', likes: 45000, comments: 1200 },
  { id: '3', title: 'Desi Hip-Hop Drop 🔥', user: '@desi_beats', likes: 8500, comments: 210 },
];

export default function App() {
  const [activeTab, setActiveTab] = useState('Home');
  const [uploadProgress, setUploadProgress] = useState(0);
  const [isUploading, setIsUploading] = useState(false);
  const [selectedFile, setSelectedFile] = useState(null);
  const [selectedFileUri, setSelectedFileUri] = useState(null);
  const [caption, setCaption] = useState('');
  const [isEditingProfile, setIsEditingProfile] = useState(false);
  const [profile, setProfile] = useState({
    name: 'Tech Creator',
    handle: '@tech_creator',
    bio: 'Welcome to my channel! Subscribe for daily coding shorts and tech reviews. 🚀💻'
  });
  const [editForm, setEditForm] = useState(profile);

  // --- HOME SCREEN ---
  const VideoItem = ({ item }) => {
    const [isLiked, setIsLiked] = useState(false);
    const [showComments, setShowComments] = useState(false);
    const [newComment, setNewComment] = useState('');
    const [commentsList, setCommentsList] = useState([
      { id: 'c1', user: 'User123', text: 'This track is fire! 🔥' },
      { id: 'c2', user: 'DesiBeats', text: 'Need this audio ASAP.' },
      { id: 'c3', user: 'MusicLover', text: 'When is the full release?' }
    ]);

    const handleShare = async () => {
      try {
        await Share.share({
          message: `Check out this amazing video: ${item.title} by ${item.user}! Watch it on the app now. 🔥`,
        });
      } catch (error) {
        Alert.alert('Error', 'Error sharing: ' + error.message);
      }
    };

    const handlePostComment = () => {
      if (newComment.trim().length > 0) {
        setCommentsList([...commentsList, {
          id: Date.now().toString(),
          user: 'You',
          text: newComment.trim(),
        }]);
        setNewComment('');
      }
    };

    return (
      <View style={styles.videoContainer}>
        <View style={styles.videoPlayerMock}>
          <Text style={styles.videoMockText}>▶</Text>
          <Text style={styles.videoMockSubText}>Playing Track...</Text>
        </View>

        <View style={styles.interactionBar}>
          <TouchableOpacity style={styles.iconButton} onPress={() => setIsLiked(!isLiked)}>
            <Text style={styles.iconText}>{isLiked ? '❤️' : '🤍'}</Text>
            <Text style={styles.iconSubText}>{isLiked ? item.likes + 1 : item.likes}</Text>
          </TouchableOpacity>
          <TouchableOpacity style={styles.iconButton} onPress={() => setShowComments(true)}>
            <Text style={styles.iconText}>💬</Text>
            <Text style={styles.iconSubText}>{item.comments + commentsList.length - 3}</Text>
          </TouchableOpacity>
          <TouchableOpacity style={styles.iconButton} onPress={handleShare}>
            <Text style={styles.iconText}>↗️</Text>
            <Text style={styles.iconSubText}>Share</Text>
          </TouchableOpacity>
        </View>

        <View style={styles.bottomInfo}>
          <Text style={styles.username}>{item.user}</Text>
          <Text style={styles.songTitle}>♫ {item.title}</Text>
        </View>

        <Modal visible={showComments} transparent={true} animationType="slide">
          <View style={styles.modalBackground}>
            <View style={styles.commentsSheet}>
              <View style={styles.modalHeader}>
                <Text style={styles.modalTitle}>Comments ({item.comments + commentsList.length - 3})</Text>
                <TouchableOpacity onPress={() => setShowComments(false)}>
                  <Text style={styles.closeText}>Close</Text>
                </TouchableOpacity>
              </View>
              <FlatList
                data={commentsList}
                keyExtractor={(c) => c.id}
                showsVerticalScrollIndicator={false}
                style={styles.commentListArea}
                renderItem={({ item: commentItem }) => (
                  <Text style={styles.commentText}>
                    <Text style={styles.commentUsername}>{commentItem.user}: </Text>
                    {commentItem.text}
                  </Text>
                )}
              />
              <View style={styles.commentInputRow}>
                <TextInput
                  style={styles.commentInput}
                  placeholder="Add a comment..."
                  placeholderTextColor="#888"
                  value={newComment}
                  onChangeText={setNewComment}
                />
                <TouchableOpacity style={styles.postCommentBtn} onPress={handlePostComment}>
                  <Text style={styles.postCommentBtnText}>Post</Text>
                </TouchableOpacity>
              </View>
            </View>
          </View>
        </Modal>
      </View>
    );
  };

  const renderHome = () => (
    <FlatList
      data={VIDEO_FEED}
      keyExtractor={item => item.id}
      pagingEnabled={true}
      showsVerticalScrollIndicator={false}
      renderItem={({ item }) => <VideoItem item={item} />}
    />
  );

  // --- UPLOAD STUDIO WITH REAL FILE PICKER ---
  const requestPermissionAndPick = async () => {
    try {
      // Request media library permission (needed on Android for storage access)
      if (Platform.OS === 'android') {
        const { status } = await MediaLibrary.requestPermissionsAsync();
        if (status === 'denied') {
          Alert.alert(
            'Permission Required',
            'Storage permission is needed to select audio files. Please enable it in your device Settings > Apps > [App Name] > Permissions.',
            [{ text: 'OK' }]
          );
          return;
        }
      }

      // Open the native file picker
      const result = await DocumentPicker.getDocumentAsync({
        type: ['audio/*', 'audio/mpeg', 'audio/mp3', 'audio/wav', 'audio/m4a', 'audio/ogg'],
        copyToCacheDirectory: true,   // copies file to app cache so it's accessible
        multiple: false,
      });

      console.log('Picker result:', JSON.stringify(result));

      // Handle result — Expo SDK 49+ uses result.assets array
      if (!result.canceled && result.assets && result.assets.length > 0) {
        const file = result.assets[0];
        setSelectedFile(file.name || 'Selected Audio File');
        setSelectedFileUri(file.uri);
        Alert.alert('✅ File Selected', `File: ${file.name}\nSize: ${file.size ? (file.size / 1024).toFixed(1) + ' KB' : 'Unknown'}`);
      
      // Fallback for older Expo SDK (result.type === 'success')
      } else if (result.type === 'success') {
        setSelectedFile(result.name || 'Selected Audio File');
        setSelectedFileUri(result.uri);
        Alert.alert('✅ File Selected', `File: ${result.name}`);
      
      } else {
        // User cancelled — do nothing silently
        console.log('User cancelled file picker');
      }

    } catch (error) {
      console.error('File picker error:', error);
      Alert.alert('Error', 'Could not open file picker: ' + error.message);
    }
  };

  const handleSimulatedUpload = () => {
    if (!selectedFile) {
      Alert.alert('No File', 'Please select an MP3/Audio file first!');
      return;
    }
    setIsUploading(true);
    setUploadProgress(0);
    let progress = 0;
    const interval = setInterval(() => {
      progress += 10;
      setUploadProgress(progress);
      if (progress >= 100) {
        clearInterval(interval);
        setTimeout(() => {
          setIsUploading(false);
          Alert.alert('🎉 Success', 'Audio Uploaded Successfully!');
          setSelectedFile(null);
          setSelectedFileUri(null);
          setCaption('');
          setActiveTab('Home');
        }, 500);
      }
    }, 300);
  };

  const renderUpload = () => (
    <ScrollView style={styles.studioContainer} keyboardShouldPersistTaps="handled">
      <Text style={styles.pageHeader}>Upload New Track</Text>

      {/* File Picker Button */}
      <TouchableOpacity style={styles.uploadBox} onPress={requestPermissionAndPick} activeOpacity={0.7}>
        <Text style={styles.uploadIcon}>{selectedFile ? '🎵' : '📁'}</Text>
        <Text style={styles.uploadText}>
          {selectedFile ? `✅ ${selectedFile}` : 'Tap to Browse & Select Audio File'}
        </Text>
        {!selectedFile && (
          <Text style={styles.uploadSubText}>Supports MP3, WAV, M4A, OGG</Text>
        )}
      </TouchableOpacity>

      {/* Clear selection button */}
      {selectedFile && (
        <TouchableOpacity
          style={styles.clearBtn}
          onPress={() => { setSelectedFile(null); setSelectedFileUri(null); }}
        >
          <Text style={styles.clearBtnText}>✕ Remove Selected File</Text>
        </TouchableOpacity>
      )}

      <TextInput
        style={styles.captionInput}
        placeholder="Write a catchy caption for your track..."
        placeholderTextColor="#888"
        multiline
        value={caption}
        onChangeText={setCaption}
      />

      {isUploading ? (
        <View style={styles.progressContainer}>
          <Text style={styles.progressText}>Uploading... {uploadProgress}%</Text>
          <View style={styles.progressBarBg}>
            <View style={[styles.progressBarFill, { width: `${uploadProgress}%` }]} />
          </View>
        </View>
      ) : (
        <TouchableOpacity
          style={[styles.publishButton, !selectedFile && styles.publishButtonDisabled]}
          onPress={handleSimulatedUpload}
          activeOpacity={selectedFile ? 0.8 : 1}
        >
          <Text style={styles.publishButtonText}>
            {selectedFile ? '🚀 Publish Track' : 'Select a file to publish'}
          </Text>
        </TouchableOpacity>
      )}
    </ScrollView>
  );

  // --- PROFILE TAB ---
  const saveProfileSettings = () => {
    setProfile(editForm);
    setIsEditingProfile(false);
  };

  const renderProfile = () => (
    <ScrollView style={styles.studioContainer}>
      <Text style={styles.pageHeader}>My Channel</Text>
      <View style={styles.profileHeader}>
        <View style={styles.avatarPlaceholder}>
          <Text style={styles.avatarText}>{profile.name.charAt(0)}</Text>
        </View>
        <View style={styles.statsContainer}>
          <View style={styles.statBox}>
            <Text style={styles.statNumber}>124</Text>
            <Text style={styles.statLabel}>Tracks</Text>
          </View>
          <View style={styles.statBox}>
            <Text style={styles.statNumber}>14.2K</Text>
            <Text style={styles.statLabel}>Subscribers</Text>
          </View>
        </View>
      </View>

      {isEditingProfile ? (
        <View style={styles.editProfileArea}>
          <Text style={styles.inputLabel}>Display Name</Text>
          <TextInput style={styles.profileInput} value={editForm.name} onChangeText={(text) => setEditForm({...editForm, name: text})} />
          <Text style={styles.inputLabel}>Channel Handle</Text>
          <TextInput style={styles.profileInput} value={editForm.handle} onChangeText={(text) => setEditForm({...editForm, handle: text})} />
          <Text style={styles.inputLabel}>Channel Bio</Text>
          <TextInput style={[styles.profileInput, { height: 80, textAlignVertical: 'top' }]} value={editForm.bio} onChangeText={(text) => setEditForm({...editForm, bio: text})} multiline />
          <TouchableOpacity style={styles.saveProfileBtn} onPress={saveProfileSettings}>
            <Text style={styles.saveProfileText}>Save Changes</Text>
          </TouchableOpacity>
          <TouchableOpacity style={styles.cancelProfileBtn} onPress={() => setIsEditingProfile(false)}>
            <Text style={styles.cancelProfileText}>Cancel</Text>
          </TouchableOpacity>
        </View>
      ) : (
        <View style={styles.viewProfileArea}>
          <Text style={styles.profileNameDisplay}>{profile.name}</Text>
          <Text style={styles.profileHandleDisplay}>{profile.handle}</Text>
          <Text style={styles.profileBioDisplay}>{profile.bio}</Text>
          <TouchableOpacity style={styles.editProfileBtn} onPress={() => setIsEditingProfile(true)}>
            <Text style={styles.editProfileText}>✏️ Customize Channel</Text>
          </TouchableOpacity>
        </View>
      )}
    </ScrollView>
  );

  // --- BOTTOM NAV ---
  const NavBar = () => (
    <View style={styles.navBar}>
      {['Home', 'Upload', 'Profile'].map((tab) => (
        <TouchableOpacity key={tab} onPress={() => setActiveTab(tab)} style={styles.navItem}>
          <Text style={[styles.navText, activeTab === tab && styles.navTextActive]}>
            {tab === 'Upload' ? '➕ Create' : tab}
          </Text>
        </TouchableOpacity>
      ))}
    </View>
  );

  return (
    <SafeAreaView style={styles.mainContainer}>
      <StatusBar barStyle="light-content" />
      <View style={styles.contentArea}>
        {activeTab === 'Home' && renderHome()}
        {activeTab === 'Upload' && renderUpload()}
        {activeTab === 'Profile' && renderProfile()}
      </View>
      <NavBar />
    </SafeAreaView>
  );
}
